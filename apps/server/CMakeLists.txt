set(STREAMDESK_BOOST_ROOT "${PROJECT_SOURCE_DIR}/build/deps/boost_1_88_0" CACHE PATH "Fixed Boost 1.88.0 header root")
if(NOT EXISTS "${STREAMDESK_BOOST_ROOT}/boost/asio.hpp")
    message(FATAL_ERROR "Boost headers missing. Run: python3 tools/fetch_boost.py (project-local download only)")
endif()
file(STRINGS "${STREAMDESK_BOOST_ROOT}/boost/version.hpp" boost_version_line REGEX "^#define BOOST_VERSION ")
if(NOT boost_version_line STREQUAL "#define BOOST_VERSION 108800")
    message(FATAL_ERROR "This build requires the recorded Boost 1.88.0 version")
endif()
add_executable(streamdesk_server main.cpp)
target_link_libraries(streamdesk_server PRIVATE StreamCore)
target_include_directories(streamdesk_server SYSTEM PRIVATE "${STREAMDESK_BOOST_ROOT}")

    set_tests_properties(streamdesk.recording PROPERTIES TIMEOUT 90)

    set_tests_properties(streamdesk.catalog PROPERTIES TIMEOUT 60)
endif()
